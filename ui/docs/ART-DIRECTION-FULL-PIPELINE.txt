#!/usr/bin/env python3
"""
Ravenstack Keep — Asset Pipeline (Aseprite CLI & Python)
Automates palette enforcement, pixel-snapping, directory organization,
and Phaser 3 spritesheet generation.
"""
import os
import sys
import shutil
import subprocess
from pathlib import Path
from PIL import Image

PALETTE_COLORS = [
    ("bg",        (11,  14,  20)),   # #0b0e14
    ("stoneDim",  (30,  34,  43)),   # #1e222b
    ("stone",     (58,  63,  75)),   # #3a3f4b
    ("stoneLive", (74,  85,  104)),  # #4a5568
    ("neon",      (45,  226, 230)),  # #2de2e6
    ("magenta",   (255, 42,  109)),  # #ff2a6d
    ("amber",     (255, 200, 87)),   # #ffc857
    ("green",     (57,  255, 20)),   # #39ff14
    ("red",       (255, 59,  59))    # #ff3b3b
]

TARGET_DIRS = {
    "base": "ui/public/art/tiles/base",
    "facades": "ui/public/art/tiles/facades",
    "chips": "ui/public/art/chips",
    "hud": "ui/public/art/hud"
}

def create_gpl_palette(filename="ravenstack_palette.gpl"):
    lines = ["GIMP Palette", "Name: Ravenstack Keep Locked Palette", "Columns: 3", "#"]
    for name, (r, g, b) in PALETTE_COLORS:
        lines.append(f"{r:3d} {g:3d} {b:3d}\t{name}")
    with open(filename, "w") as f:
        f.write("\n".join(lines) + "\n")
    return filename

def find_nearest_palette_color(rgb, allow_alpha=True):
    r, g, b = rgb[:3]
    if len(rgb) > 3 and rgb[3] < 128 and allow_alpha:
        return (0, 0, 0, 0)
    min_dist = float("inf")
    closest = PALETTE_COLORS[0][1]
    for _, (pr, pg, pb) in PALETTE_COLORS:
        dist = (r - pr)**2 + (g - pg)**2 + (b - pb)**2
        if dist < min_dist:
            min_dist = dist
            closest = (pr, pg, pb)
    if len(rgb) > 3:
        return (closest[0], closest[1], closest[2], 255)
    return closest

def quantize_and_resize(image_path, output_path, target_size=(48, 48)):
    img = Image.open(image_path).convert("RGBA")
    img_resized = img.resize(target_size, Image.Resampling.NEAREST)
    pixels = img_resized.load()
    width, height = img_resized.size
    for y in range(height):
        for x in range(width):
            pixels[x, y] = find_nearest_palette_color(pixels[x, y])
    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    img_resized.save(output_path)

def process_file(input_file, output_file, palette_file, target_size=(48, 48)):
    quantize_and_resize(input_file, output_file, target_size)
    aseprite_bin = shutil.which("aseprite")
    if aseprite_bin:
        cmd = [aseprite_bin, "-b", output_file, "--palette", palette_file, "--save-as", output_file]
        try:
            subprocess.run(cmd, check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        except subprocess.CalledProcessError as e:
            print(f"Aseprite CLI error: {e}")

def build_phaser3_spritesheet(art_root="ui/public/art"):
    aseprite_bin = shutil.which("aseprite")
    if not aseprite_bin:
        return
    tile_files = list(Path(art_root).glob("**/*.png"))
    valid_tiles = [str(f) for f in tile_files if f.name != "ravenstack_sheet.png"]
    if not valid_tiles:
        return
    sheet_png = os.path.join(art_root, "ravenstack_sheet.png")
    sheet_json = os.path.join(art_root, "ravenstack_sheet.json")
    cmd = [aseprite_bin, "-b", *valid_tiles, "--sheet-type", "packed", "--sheet", sheet_png, "--data", sheet_json, "--format", "json-hash"]
    try:
        subprocess.run(cmd, check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    except subprocess.CalledProcessError as e:
        print(f"Failed to generate spritesheet: {e}")

def main():
    for path in TARGET_DIRS.values():
        os.makedirs(path, exist_ok=True)
    gpl_file = create_gpl_palette()
    input_dir = Path("input")
    if not input_dir.exists():
        os.makedirs(input_dir, exist_ok=True)
        return
    raw_images = list(input_dir.glob("*.png")) + list(input_dir.glob("*.jpg"))
    for img_path in raw_images:
        filename = img_path.name
        if filename.startswith("chip_") or filename.startswith("prop_ether"):
            target_path = os.path.join(TARGET_DIRS["chips"], filename)
            size = (16, 16)
        elif filename.startswith("hud_") or filename.startswith("gate_") or filename.startswith("selection_") or filename.startswith("prop_scan") or filename.startswith("prop_terminal"):
            target_path = os.path.join(TARGET_DIRS["hud"], filename)
            size = (192, 64) if filename.startswith("hud_") else ((24, 24) if filename.startswith("gate_") else ((32, 32) if filename.startswith("prop_terminal") else (48, 48)))
        elif filename.startswith("facade_"):
            target_path = os.path.join(TARGET_DIRS["facades"], filename)
            size = (48, 48)
        else:
            target_path = os.path.join(TARGET_DIRS["base"], filename)
            size = (48, 48)
        process_file(str(img_path), target_path, gpl_file, target_size=size)
    build_phaser3_spritesheet()

if __name__ == "__main__":
    main()

QA & Rejection Criteria
Reject any generated image exhibiting:

Blurry edges, sub-pixel rendering, or anti-aliasing.
Rendered text, English letters, or legible numbers.
Incorrect bounding box or missing transparency padding.
Neon glow exceeding 20% of the tile surface area.
Skewed 3D camera angles or isometric perspective.
Unapproved colors outside the locked hex palette.
